\section{Theorem-like environments from \texttt{theorem.tex}}

\begin{definition}[SPD matrix]
A matrix $A\in\R^{n\times n}$ is symmetric positive definite (SPD) if
$A=A^\top$ and $\bm{x}^\top A\bm{x}>0$ for all $\bm{x}\neq 0$.
\end{definition}

\begin{assumption}
Assume $A$ is SPD.
\end{assumption}

\begin{lemma}
If $A$ is SPD, then all eigenvalues of $A$ are positive.
\end{lemma}

\begin{theorem}
If $A$ is SPD, then $\tr(A)>0$.
\end{theorem}

\begin{proof}
SPD implies eigenvalues $\lambda_i>0$. Then $\tr(A)=\sum_i \lambda_i>0$.
For general LaTeX conventions, see \textcite{lamport1994}.
\end{proof}

\begin{proposition}
If $A$ is SPD and $D=\diag(d_1,\dots,d_n)$ with all $d_i>0$, then $DAD$ is SPD.
\end{proposition}

\begin{corollary}
If $A$ is SPD, then $\trace(A^{-1})>0$.
\end{corollary}

\begin{example}
Let $A=\diag(2,1)$. Then $A$ is SPD and $\tr(A)=3$.
\end{example}
